CC = gcc
CFLAGS = -Wall -Wextra -g
TARGET = app
SRCS = main.c utils.c
OBJS = $(SRCS:.c=.o)

.PHONY: all build clean

all: build

build: $(TARGET)

$(TARGET): $(OBJS)
	$(CC) $(CFLAGS) -o $(TARGET) $(OBJS)

%.o: %.c utils.h
	$(CC) $(CFLAGS) -c $< -o $@

clean:
	rm -f $(OBJS) $(TARGET)
